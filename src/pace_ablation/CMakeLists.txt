cmake_minimum_required(VERSION 3.8)
project(pace_ablation)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(frontier_msgs REQUIRED)
find_package(fuzzy_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)

add_executable(feature_ablation_node src/feature_ablation_node.cpp)
ament_target_dependencies(feature_ablation_node
  rclcpp std_msgs frontier_msgs fuzzy_msgs tf2_ros tf2_geometry_msgs nav_msgs geometry_msgs)
install(TARGETS feature_ablation_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
